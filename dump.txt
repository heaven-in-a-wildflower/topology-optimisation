/*
Fixed Level-Set Topology Optimization with Volume Constraint
Based on: Allaire, Jouve, Toader (2004)

Key improvements:
1. Proper volume constraint with adaptive Lagrange multiplier
2. Normalized velocity field
3. Better numerical stability
*/

#include "mfem.hpp"
#include <iostream>
#include <fstream>
#include <cmath>
#include <filesystem>
#include <vector>

using namespace std;
using namespace mfem;
namespace fs = std::filesystem;

// Lamé parameters with ersatz material
class LameCoefficient : public Coefficient
{
private:
    GridFunction *psi;
    double E_solid, nu, E_void;
    bool is_lambda;

public:
    LameCoefficient(GridFunction *psi_, double E_s, double nu_, double E_v, bool lambda)
        : psi(psi_), E_solid(E_s), nu(nu_), E_void(E_v), is_lambda(lambda) {}

    virtual double Eval(ElementTransformation &T, const IntegrationPoint &ip)
    {
        double psi_val = psi->GetValue(T, ip);
        double epsilon = 0.05;
        double H = 0.5 * (1.0 + 2.0 / M_PI * atan(psi_val / epsilon));
        double E = E_solid + (E_void - E_solid) * H;

        if (is_lambda)
            return E * nu / ((1.0 + nu) * (1.0 - 2.0 * nu));
        else
            return E / (2.0 * (1.0 + nu));
    }
};

// Shape derivative velocity (Eq. 24)
class ShapeVelocity : public Coefficient
{
private:
    GridFunction *u;
    GridFunction *psi;
    double lagrange_mult;
    double E, nu;

public:
    ShapeVelocity(GridFunction *u_, GridFunction *psi_,
                  double ell, double E_, double nu_)
        : u(u_), psi(psi_), lagrange_mult(ell), E(E_), nu(nu_) {}

    void SetLagrangeMultiplier(double ell) { lagrange_mult = ell; }

    virtual double Eval(ElementTransformation &T, const IntegrationPoint &ip)
    {
        double lambda = E * nu / ((1.0 + nu) * (1.0 - 2.0 * nu));
        double mu = E / (2.0 * (1.0 + nu));

        DenseMatrix grad_u;
        u->GetVectorGradient(T, grad_u);

        double e11 = grad_u(0, 0);
        double e22 = grad_u(1, 1);
        double e12 = 0.5 * (grad_u(0, 1) + grad_u(1, 0));
        double trace_e = e11 + e22;

        double s11 = 2.0 * mu * e11 + lambda * trace_e;
        double s22 = 2.0 * mu * e22 + lambda * trace_e;
        double s12 = 2.0 * mu * e12;

        double strain_energy = s11 * e11 + s22 * e22 + 2.0 * s12 * e12;

        // v = ℓ - A·e(u):e(u)
        return lagrange_mult - strain_energy;
    }
};

// Upwind gradient
double UpwindGradient(const GridFunction &psi, double V,
                      int i, int j, int Nx, int Ny, double dx, double dy)
{
    int idx = i + j * Nx;
    double psi_c = psi(idx);

    double Dxp = (i < Nx - 1) ? (psi((i + 1) + j * Nx) - psi_c) / dx : 0;
    double Dxm = (i > 0) ? (psi_c - psi((i - 1) + j * Nx)) / dx : 0;
    double Dyp = (j < Ny - 1) ? (psi(i + (j + 1) * Nx) - psi_c) / dy : 0;
    double Dym = (j > 0) ? (psi_c - psi(i + (j - 1) * Nx)) / dy : 0;

    double grad_x_sq, grad_y_sq;

    if (V > 0)
    {
        grad_x_sq = max(pow(max(Dxm, 0.0), 2), pow(min(Dxp, 0.0), 2));
        grad_y_sq = max(pow(max(Dym, 0.0), 2), pow(min(Dyp, 0.0), 2));
    }
    else
    {
        grad_x_sq = max(pow(min(Dxm, 0.0), 2), pow(max(Dxp, 0.0), 2));
        grad_y_sq = max(pow(min(Dym, 0.0), 2), pow(max(Dyp, 0.0), 2));
    }

    return sqrt(grad_x_sq + grad_y_sq);
}

int main(int argc, char *argv[])
{
    MPI_Init(&argc, &argv);

    // Parameters
    const double Lx = 2.0, Ly = 1.0;
    const int Nx = 160, Ny = 80;

    const double E = 1.0;
    const double nu = 0.3;
    const double E_void = 1e-3;

    const double target_volume_fraction = 0.4; // Target 40% material
    double lagrange_mult = 1.0;                // Will be adapted

    const int max_iter = 100;
    const int hj_steps = 20; // More steps per iteration
    const int reinit_freq = 5;

    cout << "Level-Set Topology Optimization - 2D Cantilever\n";
    cout << "Target volume fraction: " << target_volume_fraction << "\n\n";

    Mesh mesh = Mesh::MakeCartesian2D(Nx, Ny, Element::QUADRILATERAL,
                                      false, Lx, Ly, false);

    H1_FECollection fec(1, 2);
    FiniteElementSpace fes_H1(&mesh, &fec);
    FiniteElementSpace fes_vec(&mesh, &fec, 2);

    cout << "DOFs: " << fes_H1.GetTrueVSize() << " (scalar), "
         << fes_vec.GetTrueVSize() << " (vector)\n\n";

    // Boundary conditions
    Array<int> ess_bdr(mesh.bdr_attributes.Max());
    ess_bdr = 0;
    ess_bdr[0] = 1;

    Array<int> ess_tdof;
    fes_vec.GetEssentialTrueDofs(ess_bdr, ess_tdof);

    // Initialize level-set with holes to provide initial topology
    GridFunction psi(&fes_H1);

    // Create a grid of holes (checkerboard-like pattern)
    int holes_x = 8; // Number of holes in x-direction
    int holes_y = 4; // Number of holes in y-direction
    double hole_radius = 0.08;

    for (int i = 0; i < psi.Size(); i++)
    {
        const double *coord = mesh.GetVertex(i);
        double x = coord[0], y = coord[1];

        // Start with material everywhere
        double min_dist = 1e10;

        // Create grid of circular holes
        for (int hx = 0; hx < holes_x; hx++)
        {
            for (int hy = 0; hy < holes_y; hy++)
            {
                double cx = (hx + 0.5) * Lx / holes_x;
                double cy = (hy + 0.5) * Ly / holes_y;
                double dist = sqrt(pow(x - cx, 2) + pow(y - cy, 2)) - hole_radius;
                min_dist = min(min_dist, dist);
            }
        }

        // psi < 0 = material, psi > 0 = void
        psi(i) = min_dist;
    }

    // Calculate initial volume
    int initial_material = 0;
    for (int i = 0; i < psi.Size(); i++)
        if (psi(i) < 0)
            initial_material++;
    double initial_vol_frac = (double)initial_material / psi.Size();
    cout << "Initial volume fraction: " << initial_vol_frac << "\n";

    // Grid parameters
    double dx = Lx / Nx;
    double dy = Ly / Ny;
    double dt_hj = 0.2 * min(dx, dy);

    GridFunction u(&fes_vec);
    u = 0.0;

    vector<double> compliance_history;
    vector<double> volume_history;

    for (int iter = 0; iter < max_iter; iter++)
    {
        cout << "\n=== Iteration " << iter + 1 << " ===\n";

        // Solve elasticity
        LameCoefficient lambda_c(&psi, E, nu, E_void, true);
        LameCoefficient mu_c(&psi, E, nu, E_void, false);

        BilinearForm a(&fes_vec);
        a.AddDomainIntegrator(new ElasticityIntegrator(lambda_c, mu_c));
        a.Assemble();

        // Point load
        LinearForm b(&fes_vec);
        b = 0.0;

        double load_x = Lx, load_y = Ly / 2.0;
        double min_dist = 1e10;
        int load_vertex = -1;

        for (int i = 0; i < mesh.GetNV(); i++)
        {
            const double *coord = mesh.GetVertex(i);
            double dist = sqrt(pow(coord[0] - load_x, 2) +
                               pow(coord[1] - load_y, 2));
            if (dist < min_dist)
            {
                min_dist = dist;
                load_vertex = i;
            }
        }

        if (load_vertex >= 0)
        {
            Array<int> vdofs;
            fes_vec.GetVertexVDofs(load_vertex, vdofs);
            if (vdofs.Size() >= 2)
            {
                int y_dof = vdofs[1];
                if (y_dof < 0)
                    y_dof = -1 - y_dof;
                Vector &b_vec = b;
                b_vec(y_dof) = -1.0;
            }
        }

        // Solve
        OperatorPtr A;
        Vector B, X;
        a.FormLinearSystem(ess_tdof, u, b, A, X, B);

        GSSmoother M((SparseMatrix &)(*A));
        PCG(*A, M, B, X, 0, 1000, 1e-10, 0.0);

        a.RecoverFEMSolution(X, b, u);

        // Compute compliance
        double compliance = b * u;

        // Compute volume fraction
        int material_count = 0;
        for (int i = 0; i < psi.Size(); i++)
            if (psi(i) < 0)
                material_count++;

        double volume_fraction = (double)material_count / psi.Size();

        compliance_history.push_back(compliance);
        volume_history.push_back(volume_fraction);

        cout << "Compliance: " << compliance << "\n";
        cout << "Volume fraction: " << volume_fraction << "\n";

        // Adaptive Lagrange multiplier to enforce volume constraint
        double volume_error = volume_fraction - target_volume_fraction;

        if (iter > 0)
        {
            // Aggressive adaptation based on volume error
            double alpha = 2.0 * compliance; // Scale with compliance
            lagrange_mult += alpha * volume_error;
            lagrange_mult = max(0.01, min(lagrange_mult, 1000.0)); // Wider bounds
        }

        cout << "Volume error: " << volume_error << "\n";
        cout << "Lagrange multiplier: " << lagrange_mult << "\n";

        // Compute velocity
        ShapeVelocity velocity(&u, &psi, lagrange_mult, E, nu);
        GridFunction vel(&fes_H1);
        vel.ProjectCoefficient(velocity);

        // Check velocity statistics BEFORE normalization
        double min_vel_raw = 1e10, max_vel_raw = -1e10, avg_vel_raw = 0.0;
        for (int i = 0; i < vel.Size(); i++)
        {
            min_vel_raw = min(min_vel_raw, vel(i));
            max_vel_raw = max(max_vel_raw, vel(i));
            avg_vel_raw += vel(i);
        }
        avg_vel_raw /= vel.Size();

        cout << "Raw velocity: min=" << min_vel_raw
             << ", max=" << max_vel_raw
             << ", avg=" << avg_vel_raw << "\n";

        // Normalize velocity to prevent instability
        double max_vel = 0.0;
        for (int i = 0; i < vel.Size(); i++)
            max_vel = max(max_vel, fabs(vel(i)));

        if (max_vel > 1e-8)
        {
            double scale = 1.0 / max_vel;
            for (int i = 0; i < vel.Size(); i++)
                vel(i) *= scale;
        }
        else
        {
            cout << "WARNING: Velocity field is essentially zero!\n";
        }

        cout << "Velocity normalization factor: " << max_vel << "\n";

        // Evolve level-set
        for (int step = 0; step < hj_steps; step++)
        {
            GridFunction psi_new = psi;

            for (int j = 0; j < Ny; j++)
            {
                for (int i = 0; i < Nx; i++)
                {
                    int idx = i + j * Nx;
                    double v = vel(idx);
                    double grad = UpwindGradient(psi, v, i, j, Nx, Ny, dx, dy);
                    psi_new(idx) = psi(idx) - dt_hj * v * grad;
                }
            }

            psi = psi_new;
        }

        // Reinitialize
        if ((iter + 1) % reinit_freq == 0)
        {
            cout << "Reinitializing level-set...\n";
            GridFunction psi0 = psi;
            double dt_r = 0.1 * min(dx, dy);

            for (int r = 0; r < 5; r++)
            {
                GridFunction psi_temp = psi;
                for (int j = 0; j < Ny; j++)
                {
                    for (int i = 0; i < Nx; i++)
                    {
                        int idx = i + j * Nx;
                        double sign = psi0(idx) / sqrt(psi0(idx) * psi0(idx) + dx * dx);
                        double grad = UpwindGradient(psi, sign, i, j, Nx, Ny, dx, dy);
                        psi_temp(idx) = psi(idx) - dt_r * sign * (grad - 1.0);
                    }
                }
                psi = psi_temp;
            }
        }

        // Convergence check
        if (iter > 10)
        {
            double comp_change = fabs(compliance_history[iter] -
                                      compliance_history[iter - 5]) /
                                 compliance_history[iter];
            double vol_error = fabs(volume_fraction - target_volume_fraction);

            cout << "Compliance change (5 iter): " << comp_change << "\n";
            cout << "Volume error: " << vol_error << "\n";

            if (comp_change < 0.01 && vol_error < 0.02)
            {
                cout << "\n✓ Converged!\n";
                break;
            }
        }
    }

    // Save results
    fs::create_directories("levelset_results");

    ofstream mesh_ofs("levelset_results/cantilever.mesh");
    mesh.Print(mesh_ofs);

    GridFunction density(&fes_H1);
    for (int i = 0; i < density.Size(); i++)
        density(i) = (psi(i) < 0) ? 1.0 : 0.0;

    ofstream density_ofs("levelset_results/density_final.gf");
    density.Save(density_ofs);

    ofstream disp_ofs("levelset_results/displacement_final.gf");
    u.Save(disp_ofs);

    // Save convergence history
    ofstream hist_ofs("levelset_results/history.txt");
    hist_ofs << "# Iteration Compliance VolumeF raction\n";
    for (size_t i = 0; i < compliance_history.size(); i++)
        hist_ofs << i + 1 << " " << compliance_history[i] << " "
                 << volume_history[i] << "\n";

    cout << "\n=== Optimization Complete ===\n";
    cout << "Final compliance: " << compliance_history.back() << "\n";
    cout << "Final volume fraction: " << volume_history.back() << "\n";
    cout << "Results saved to levelset_results/\n";

    MPI_Finalize();
    return 0;
}
